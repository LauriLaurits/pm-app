### Task 1: Reports data layer (queries + compute + tests)

**Files:** rework `src/app/(app)/reports/{queries.ts, compute.ts, types.ts}`; Test: `tests/reports-compute.test.ts`.

**Interfaces (later tasks consume EXACTLY):**
```ts
export type ReportsWindow = { months: 3 | 6 | 12; startISO: string; endISO: string; prevStartISO: string; label: string };
export function reportsWindow(months: number, todayISO: string): ReportsWindow; // label "1 Feb – 31 Jul 2026"
export type TrendDelta = { pct: number; direction: "up" | "down" | "flat" } | null; // null = no prior data
export function trendDelta(current: number, previous: number): TrendDelta;
export type ReportsKpis = {
  totalHours: number; hoursDelta: TrendDelta;
  totalInvoiced: number | null; invoicedDelta: TrendDelta;   // null without budget visibility
  totalCost: number | null; costDelta: TrendDelta;           // null without finance visibility
  budgetRemaining: number | null;                            // current-state, no delta
  avgUtilization: number; availableInfo: string;             // current-state, no delta
};
export type MonthlyHoursPoint = { month: string; billable: number; nonBillable: number };
export function cumulative(points: MonthlyHoursPoint[]): MonthlyHoursPoint[];
export type ProjectHoursSlice = { id: string | null; name: string; hours: number; pct: number }; // top 5 + "Other projects"
export type UtilizationRow = { id: string; name: string; invoiced: number | null; budget: number | null; consumptionPct: number | null; remaining: number | null };
export type CapacityRow = { id: string; name: string; avatarUrl: string | null; pct: number; allocatedHours: number; capacityHours: number };
export type MonthlyFinancePoint = { month: string; invoiced: number; cost: number | null };
export type PerformanceRow = {
  id: string; name: string; clientName: string | null; budget: number | null; invoiced: number | null;
  invoicedPct: number | null; cost: number | null; costPct: number | null; remaining: number | null;
  marginPct: number | null; hoursLogged: number; billablePct: number | null; status: BudgetStatus | "no_budget";
};
```

One parallel wave.

- vitest (write FIRST, fail, then implement): reportsWindow boundaries + label; trendDelta (up/down/flat/zero-prev→null); cumulative(); top-5+Other slicing incl. pct sum ≈ 100; performance rollup with a gated-null fixture (invoiced null stays null, no 0-coercion).
- Commit: `feat: reports v2 data layer`
